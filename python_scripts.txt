import random

def generatePass(parLen = 8, parDict = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"):
    strPass = ""
    for i in range(parLen):
        strPass = strPass + random.choice(parDict)
        
    return strPass

def limit_text(text_input, word_count):
    words = text_input.split()
    
    if len(words) > word_count:
        words_lim = words[:word_count]
        return ' '.join(words_lim) + "..."
    else:
        return ' '.join(words)
        

print(limit_text("qwe rtya sdff  ghzxc   vbn", 4))

def divide(a, b):
    if b == 0:
        raise ValueError("Делитель не может быть равен нулю")
    return a / b

try:
    a = int(input("Введите делимое: "))
    b = int(input("Введите делитель: "))
    print(divide(a, b))
    
except ZeroDivisionError:
    # Это исключение не возникнет, так как мы его обработали в функции
    # и вызываем другое исключение
    print("Ошибка деления на ноль")
except ValueError:
    # Это исключение возникнет
    print("Ошибка значения")
